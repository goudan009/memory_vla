## 启动评估

```bash
# 默认：baseline，GPU1，单环境，Task21，10个完整Episode，保存64倍速视频
./evaluation/scripts/run_groot_eval.sh

# baseline：GPU0，双环境各5回合
  ./evaluation/scripts/run_groot_eval.sh \
    --version V_1 --group baseline \
    --mode single2 --gpu-index 0 --task 21 --seed 0 \
    --policy-port 5560 --video-steps-per-render 16

# experment：GPU1，双环境各5回合
  ./evaluation/scripts/run_groot_eval.sh \
    --version V_1 --group experment_1 \
    --mode single2 --gpu-index 1 --task 21 --seed 0 \
    --policy-port 5561 --video-steps-per-render 16

# V0baseline评估
  ./evaluation/scripts/run_groot_eval.sh \
    --version V_0 --group baseline \
    --mode single2 --gpu-index 1 --task 40 --seed 0 \
    --policy-port 5556 --video-steps-per-render 16

# 参数说明（按需添加或修改）
  --version V_1                 # 版本目录，支持自定义名称
  --group baseline              # 分组：baseline、experment_1等，支持自定义
  --model-name groot            # 文件夹前缀：groot /grootmeory，不切换模型
  --run-tag memory_v1           # 运行备注，不改变文件夹名称
  --task 22                     # 任务：21 / 22 / 26
  --mode single2                # single1 / single2 / dual1 /dual2
  --gpu-index 1                 # GPU：0 / 1；双卡模式不要加
  --episodes-per-gpu 10         # 每张卡的总回合数；正式默认单卡10、双卡每卡5
  --instances-gpu0 0 1 2 3 4    # GPU0指定实例
  --instances-gpu1 5 6 7 8 9    # GPU1指定实例；双卡实例不能重叠
  --policy-host 172.28.62.17    # 推理服务地址
  --policy-port 5556            # 推理服务端口，决定连接哪个模型
  --seed 0                      # 基准种子；填random则随机生成并记录
  --seed-mode sequence          # sequence递增 / fixed固定 /random可复现随机
  --preset official-eval        # 正式评估，默认
  --preset fast-debug           # 快速调试，不计正式成绩；与上一项二选一
  --debug-max-steps 1200        # 仅fast-debug可用，底层动作步数上限
  --record-video                # 开启录像，默认
  --no-record-video             # 关闭录像，与上一项二选一
  --video-steps-per-render 96   # 96为64倍速；48为32倍速
  --resume-from 运行目录         # 从回合边界续跑，继承原配置
  --wait-for-gpu                # 显存不满足启动条件时等待；续跑默认等待
  --gpu-wait-interval 30        # 等待时每30秒检查一次
  --diagnostics                 # 保留额外诊断数据
  --appdata-gpu0 缓存目录        # GPU0缓存路径，通常不用改
  --appdata-gpu1 缓存目录        # GPU1缓存路径，双卡不能共用同一目录
  --dry-run                     # 只检查并打印命令，不启动评估
  --help                        # 查看帮助
```
